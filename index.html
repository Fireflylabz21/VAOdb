function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (Array.isArray(data)) {
    data.forEach(function(d) {
      sheet.appendRow([d.state, d.metro, d.corridor, d.category, d.firmName, d.repName, d.mobile, d.altMobile, d.email, d.zoning, d.parcelSize, d.reraId, d.verification]);
    });
  } else {
    sheet.appendRow([data.state, data.metro, data.corridor, data.category, data.firmName, data.repName, data.mobile, data.altMobile, data.email, data.zoning, data.parcelSize, data.reraId, data.verification]);
  }
  return ContentService.createTextOutput("Success");
}
